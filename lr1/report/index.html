    Задание 1. Обмен сообщениями по UDP

    Описание

    Клиент должен отправить серверу сообщение Hello, server. Сервер выводит полученное сообщение и отвечает строкой Hello, client, которую выводит клиент.

    UDP не устанавливает постоянное соединение. Для передачи датаграмм используются методы sendto() и recvfrom(), а сокет создаётся с типом SOCK_DGRAM.

    Сервер

    import socket


HOST = "127.0.0.1"
PORT = 5000
BUFFER_SIZE = 1024


def main() -> None:
    with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as server_socket:
        server_socket.bind((HOST, PORT))
        print(f"UDP server is listening on {HOST}:{PORT}")

        data, client_address = server_socket.recvfrom(BUFFER_SIZE)
        message = data.decode("utf-8")
        print(f"Received from client: {message}")

        response = "Hello, client"
        server_socket.sendto(response.encode("utf-8"), client_address)
        print(f"Sent to client: {response}")


if __name__ == "__main__":
    main()

    Клиент

    import socket


SERVER_ADDRESS = ("127.0.0.1", 5000)
BUFFER_SIZE = 1024


def main() -> None:
    with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as client_socket:
        message = "Hello, server"
        client_socket.sendto(message.encode("utf-8"), SERVER_ADDRESS)
        print(f"Sent to server: {message}")

        data, _ = client_socket.recvfrom(BUFFER_SIZE)
        response = data.decode("utf-8")

    Задание 2. Вычисления через TCP

    Описание

    Для варианта 2 требуется решить квадратное уравнение:

    ax² + bx + c = 0

    Клиент вводит коэффициенты a, b, c и передаёт их серверу. Сервер вычисляет дискриминант D = b² − 4ac, определяет количество действительных корней и возвращает результат.

    TCP устанавливает соединение и гарантирует доставку данных в правильном порядке. Сокет создаётся с типом SOCK_STREAM; сервер использует listen() и accept(), клиент — connect().

    Сервер

    import math
import socket


HOST = "127.0.0.1"
PORT = 5001
BUFFER_SIZE = 1024


def solve_quadratic_equation(a: float, b: float, c: float) -> str:
    if a == 0:
        return "Ошибка: коэффициент a не должен быть равен нулю"

    discriminant = b**2 - 4 * a * c

    if discriminant < 0:
        return f"Дискриминант D = {discriminant:g}. Действительных корней нет"

    if math.isclose(discriminant, 0.0):
        root = -b / (2 * a)
        return f"Дискриминант D = 0. Один корень: x = {root:g}"

    first_root = (-b + math.sqrt(discriminant)) / (2 * a)
    second_root = (-b - math.sqrt(discriminant)) / (2 * a)
    return (
        f"Дискриминант D = {discriminant:g}. "
        f"Два корня: x1 = {first_root:g}, x2 = {second_root:g}"
    )


def main() -> None:
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as server_socket:
        server_socket.bind((HOST, PORT))
        server_socket.listen(1)
        print(f"TCP server is listening on {HOST}:{PORT}")

        connection, client_address = server_socket.accept()
        with connection:
            print(f"Client connected: {client_address}")
            request = connection.recv(BUFFER_SIZE).decode("utf-8")
            print(f"Received coefficients: {request}")

            try:
                a_text, b_text, c_text = request.split(";")
                a = float(a_text)
                b = float(b_text)
                c = float(c_text)
                response = solve_quadratic_equation(a, b, c)
            except (ValueError, TypeError):
                response = "Ошибка: необходимо передать три числа в формате a;b;c"

            connection.sendall(response.encode("utf-8"))
            print(f"Sent result: {response}")


if __name__ == "__main__":
    main()

    Задание 3. Раздача HTML-страницы по HTTP

    Описание

    Сервер должен загрузить страницу из файла index.html и вернуть её клиенту как HTTP-ответ. Ответ формируется вручную: строка статуса, заголовки, пустая строка и тело страницы.

    Сервер

    import socket
from pathlib import Path


HOST = "127.0.0.1"
PORT = 8080
BUFFER_SIZE = 1024
HTML_FILE = Path(__file__).with_name("index.html")


def build_http_response() -> bytes:
    body = HTML_FILE.read_bytes()
    headers = (
        "HTTP/1.1 200 OK\r\n"
        "Content-Type: text/html; charset=utf-8\r\n"
        f"Content-Length: {len(body)}\r\n"
        "Connection: close\r\n"
        "\r\n"
    )
    return headers.encode("utf-8") + body


def main() -> None:
    response = build_http_response()

    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as server_socket:
        server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        server_socket.bind((HOST, PORT))
        server_socket.listen(5)
        print(f"HTTP server is running at http://{HOST}:{PORT}")

        try:
            while True:
                connection, client_address = server_socket.accept()
                with connection:
                    request = connection.recv(BUFFER_SIZE).decode(
                        "utf-8", errors="replace"
                    )
                    if request:
                        request_line = request.splitlines()[0]
                        print(f"{client_address}: {request_line}")
                        connection.sendall(response)
        except KeyboardInterrupt:
            print("\nHTTP server stopped")


if __name__ == "__main__":
    main()

    Задание 4. Многопользовательский чат

    Описание

    Реализован многопользовательский TCP-чат. Один и тот же клиентский файл запускается каждым пользователем. Пользователь вводит имя, отправляет сообщения и выходит командой /exit.

    Сервер хранит активные подключения и создаёт отдельный поток для каждого клиента с помощью threading.Thread. Сообщения рассылаются всем участникам, кроме отправителя. Доступ к общему словарю клиентов защищён блокировкой threading.Lock.

    Сервер

    import socket
import threading


HOST = "127.0.0.1"
PORT = 5002

clients: dict[socket.socket, str] = {}
clients_lock = threading.Lock()


def send_message(connection: socket.socket, message: str) -> None:
    connection.sendall(f"{message}\n".encode("utf-8"))


def disconnect_client(connection: socket.socket) -> str | None:
    with clients_lock:
        username = clients.pop(connection, None)

    try:
        connection.shutdown(socket.SHUT_RDWR)
    except OSError:
        pass
    connection.close()
    return username


def broadcast(message: str, sender: socket.socket | None = None) -> None:
    with clients_lock:
        recipients = [
            connection for connection in clients if connection is not sender
        ]

    for connection in recipients:
        try:
            send_message(connection, message)
        except OSError:
            disconnect_client(connection)


def handle_client(connection: socket.socket, address: tuple[str, int]) -> None:
    username: str | None = None

    try:
        reader = connection.makefile("r", encoding="utf-8")
        username = reader.readline().strip()
        if not username:
            return

        with clients_lock:
            clients[connection] = username

        print(f"{username} connected from {address}")
        send_message(connection, "Вы подключились к чату. Команда /exit — выход.")
        broadcast(f"* {username} вошёл в чат", sender=connection)

        for line in reader:
            message = line.rstrip("\n")
            if message == "/exit":
                break
            if message:
                print(f"{username}: {message}")
                broadcast(f"{username}: {message}", sender=connection)
    except (ConnectionError, OSError, UnicodeError):
        pass
    finally:
        disconnected_username = disconnect_client(connection)
        if disconnected_username:
            print(f"{disconnected_username} left the chat")
            broadcast(f"* {disconnected_username} вышел из чата")


def main() -> None:
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as server_socket:
        server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        server_socket.bind((HOST, PORT))
        server_socket.listen()
        print(f"Chat server is running on {HOST}:{PORT}")

        try:
            while True:
                connection, address = server_socket.accept()
                threading.Thread(
                    target=handle_client,
                    args=(connection, address),
                    daemon=True,
                ).start()
        except KeyboardInterrupt:
            print("\nChat server stopped")
        finally:
            with clients_lock:
                active_connections = list(clients)
            for connection in active_connections:
                disconnect_client(connection)


if __name__ == "__main__":
    main()

    Клиент

    import socket
import threading


SERVER_ADDRESS = ("127.0.0.1", 5002)


def receive_messages(client_socket: socket.socket) -> None:
    try:
        reader = client_socket.makefile("r", encoding="utf-8")
        for line in reader:
            print(f"\r{line.rstrip()}\n> ", end="", flush=True)
    except (ConnectionError, OSError, UnicodeError):
        pass


def main() -> None:
    username = input("Введите имя: ").strip()
    while not username:
        username = input("Имя не может быть пустым. Введите имя: ").strip()

    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as client_socket:
        client_socket.connect(SERVER_ADDRESS)
        client_socket.sendall(f"{username}\n".encode("utf-8"))

        receiver = threading.Thread(
            target=receive_messages,
            args=(client_socket,),
            daemon=True,
        )
        receiver.start()

        try:
            while True:
                message = input("> ")
                client_socket.sendall(f"{message}\n".encode("utf-8"))
                if message == "/exit":
                    break
        except (EOFError, KeyboardInterrupt, OSError):
            try:
                client_socket.sendall(b"/exit\n")
            except OSError:
                pass

        receiver.join(timeout=1)

    print("Вы вышли из чата")


if __name__ == "__main__":
    main()

    Задание 5. Веб-сервер GET/POST

    Описание

    Сервер должен вручную обрабатывать HTTP-запросы GET и POST. Через форму принимаются название дисциплины и оценка. Оценки группируются по дисциплине в структуре вида:

    {
    "Математика": ["5", "4"],
    "Физика": ["3"],
}

    GET / возвращает HTML-страницу с формой и журналом, а POST / добавляет новую оценку.

    Сервер

    import socket
from html import escape
from urllib.parse import parse_qs


HOST = "127.0.0.1"
PORT = 8081
BUFFER_SIZE = 1024

grades: dict[str, list[str]] = {}


def read_http_request(connection: socket.socket) -> tuple[str, str, bytes]:
    data = b""

    while b"\r\n\r\n" not in data:
        chunk = connection.recv(BUFFER_SIZE)
        if not chunk:
            raise ConnectionError("Клиент отключился до отправки заголовков")
        data += chunk

    headers_data, _, body = data.partition(b"\r\n\r\n")
    header_lines = headers_data.decode("iso-8859-1").split("\r\n")
    method, path, _ = header_lines[0].split(" ", 2)

    headers: dict[str, str] = {}
    for line in header_lines[1:]:
        if ":" in line:
            name, value = line.split(":", 1)
            headers[name.lower()] = value.strip()

    content_length = int(headers.get("content-length", "0"))
    while len(body) < content_length:
        chunk = connection.recv(BUFFER_SIZE)
        if not chunk:
            break
        body += chunk

    return method, path.partition("?")[0], body[:content_length]


def make_http_response(status: str, body: str) -> bytes:
    body_bytes = body.encode("utf-8")
    headers = (
        f"HTTP/1.1 {status}\r\n"
        "Content-Type: text/html; charset=utf-8\r\n"
        f"Content-Length: {len(body_bytes)}\r\n"
        "Connection: close\r\n"
        "\r\n"
    )
    return headers.encode("utf-8") + body_bytes


def build_page(message: str = "") -> str:
    rows = ""
    for discipline, discipline_grades in sorted(grades.items()):
        marks = ", ".join(escape(mark) for mark in discipline_grades)
        rows += (
            "<tr>"
            f"<td>{escape(discipline)}</td>"
            f"<td>{marks}</td>"
            f"<td>{len(discipline_grades)}</td>"
            "</tr>"
        )

    if rows:
        journal = (
            "<table>"
            "<thead><tr><th>Дисциплина</th><th>Оценки</th><th>Количество</th></tr></thead>"
            f"<tbody>{rows}</tbody>"
            "</table>"
        )
    else:
        journal = "<p>Оценок пока нет.</p>"

    message_html = f'<p class="message">{escape(message)}</p>' if message else ""

    return f"""<!doctype html>
<html lang="ru">
  <head>
    <meta charset="utf-8">
    <meta name="viewport" content="width=device-width, initial-scale=1">
    <title>Журнал оценок</title>
    <style>
      body {{ max-width: 760px; margin: 40px auto; padding: 0 20px; font-family: sans-serif; }}
      form {{ display: grid; gap: 12px; max-width: 360px; }}
      input, button {{ padding: 8px; font: inherit; }}
      table {{ width: 100%; margin-top: 24px; border-collapse: collapse; }}
      th, td {{ padding: 10px; border: 1px solid #999; text-align: left; }}
      .message {{ color: #176b32; }}
    </style>
  </head>
  <body>
    <h1>Журнал оценок</h1>
    {message_html}
    <form method="post" action="/">
      <label>Дисциплина <input name="discipline" required></label>
      <label>Оценка <input name="grade" required></label>
      <button type="submit">Добавить оценку</button>
    </form>
    <h2>Все оценки</h2>
    {journal}
  </body>
</html>"""


def handle_request(method: str, path: str, body: bytes) -> bytes:
    if path != "/":
        return make_http_response(
            "404 Not Found",
            '<h1>404 — страница не найдена</h1><p><a href="/">На главную</a></p>',
        )

    if method == "GET":
        return make_http_response("200 OK", build_page())

    if method == "POST":
        fields = parse_qs(body.decode("utf-8"))
        discipline = fields.get("discipline", [""])[0].strip()
        grade = fields.get("grade", [""])[0].strip()

        if not discipline or not grade:
            return make_http_response(
                "400 Bad Request",
                build_page("Необходимо заполнить оба поля."),
            )

        grades.setdefault(discipline, []).append(grade)
        return make_http_response(
            "200 OK",
            build_page(f"Добавлена оценка {grade} по дисциплине «{discipline}»."),
        )

    return make_http_response(
        "405 Method Not Allowed",
        "<h1>Метод не поддерживается</h1>",
    )


def main() -> None:
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as server_socket:
        server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        server_socket.bind((HOST, PORT))
        server_socket.listen(5)
        print(f"Grade server is running at http://{HOST}:{PORT}")

        try:
            while True:
                connection, address = server_socket.accept()
                with connection:
                    try:
                        method, path, body = read_http_request(connection)
                        print(f"{address}: {method} {path}")
                        response = handle_request(method, path, body)
                        connection.sendall(response)
                    except (ConnectionError, UnicodeError, ValueError) as error:
                        print(f"{address}: request error: {error}")
        except KeyboardInterrupt:
            print("\nGrade server stopped")


if __name__ == "__main__":
    main()
